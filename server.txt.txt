const express = require('express');
const cors = require('cors');
const app = express();

app.use(cors());

// YAHAN APNI VALUES DAALEIN
const FASTN_API_KEY = 'YAHAN_APNI_API_KEY_PASTE_KAREIN'; 
const FASTN_ORG_ID = 'YAHAN_CUSTOMER_UUID_PASTE_KAREIN'; // Jo abhi copy ki hai

app.get('/api/get-token', async (req, res) => {
  try {
    const response = await fetch('https://live.gcp.fastn.ai/api/v1/embed/token', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${FASTN_API_KEY}`,
        'Content-Type': 'application/json',
        'X-fastn-Test-Mode': 'true' // Test key ke liye zaroori hai
      },
      body: JSON.stringify({
        endOrgId: FASTN_ORG_ID,
        userEmail: 'tenanta@test.com',
        userName: 'Tenant A'
      })
    });
    
    const data = await response.json();
    // Token nikal kar frontend ko bhejein
    res.json({ token: data.data.token }); 
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

app.listen(3000, () => {
  console.log('Server running on http://localhost:3000');
});
